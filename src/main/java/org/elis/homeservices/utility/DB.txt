CREATE DATABASE home_services_db;

USE home_services_db;

CREATE TABLE citta (
	id int unsigned primary key auto_increment,
	nome varchar(30) not null
);

CREATE TABLE utente (
	id int unsigned primary key auto_increment,
	nome varchar(30) not null,
	cognome varchar(30) not null,
	email varchar(100) not null unique,
	password varchar(16) not null,
	dataNascita date not null,
	codiceFiscale varchar(16) unique,
	ruolo tinyint not null,
	tariffa decimal(6,2),
	id_citta int unsigned not null,
	foreign key (id_citta) references citta(id),

	CONSTRAINT valore_ruolo CHECK(ruolo>=0 AND ruolo<=2)
);

CREATE TABLE segnalazione (
	id int unsigned primary key auto_increment,
	motivazione text(500) not null,
	sanzionato boolean not null default false,
	id_utente_segnalato int unsigned not null,
	id_utente_segnalante int unsigned not null,
	foreign key (id_utente_segnalato) references utente(id),
	foreign key (id_utente_segnalante) references utente(id)
);

CREATE TABLE disponibilita (
	id int unsigned primary key auto_increment,
	data date not null,
	da time,
	a time,
	id_utente int unsigned not null,
	foreign key (id_utente) references utente(id)
);

CREATE TABLE veicolo (
	id int unsigned primary key auto_increment,
	nome varchar(30) not null
);

CREATE TABLE utente_veicolo (
	id int unsigned primary key auto_increment,
	id_utente int unsigned not null,
	id_veicolo int unsigned not null,
	foreign key (id_utente) references utente(id),
	foreign key (id_veicolo) references veicolo(id)
);

CREATE TABLE immagine (
	id int unsigned primary key auto_increment,
	nome varchar(30),
	percorso varchar(150) not null unique,
	isFotoProfilo boolean not null,
	id_utente int unsigned not null,
	foreign key (id_utente) references utente(id)
);

CREATE TABLE recensione (
	id int unsigned primary key auto_increment,
	voto tinyint not null,
	descrizione text(500),
	data date not null,
	segnalata boolean not null default false,
	id_utente_scrive int unsigned not null,
	id_utente_riceve int unsigned not null,
	foreign key (id_utente_scrive) references utente(id),
	foreign key (id_utente_riceve) references utente(id),

	CONSTRAINT recensione_stesso_id_utente CHECK(id_utente_scrive != id_utente_riceve)
);

CREATE TABLE professione (
	id int unsigned primary key auto_increment,
	nome varchar(30) not null
);

CREATE TABLE utente_professione (
	id int unsigned primary key auto_increment,
	id_utente int unsigned not null,
	id_professione int unsigned not null,
	foreign key (id_utente) references utente(id),
	foreign key (id_professione) references professione(id)
);

CREATE TABLE richiesta (
	id int unsigned primary key auto_increment,
	descrizione text(1000) not null,
	data date not null,
	da time not null,
	a time not null,
	indirizzo varchar(100) not null,
	stato tinyint not null,
	id_recensione int unsigned,
	id_utente_richiede int unsigned not null,
	id_utente_riceve int unsigned not null,
	id_professione int unsigned not null,
	foreign key (id_recensione) references recensione(id),
	foreign key (id_utente_richiede) references utente(id),
	foreign key (id_utente_riceve) references utente(id),
	foreign key (id_professione) references professione(id),

	CONSTRAINT richiesta_stesso_id_utente CHECK(id_utente_richiede != id_utente_riceve),
	CONSTRAINT valore_stato CHECK(stato>=0 AND stato <=2)
);

CREATE TABLE formpro (
	id int unsigned primary key auto_increment,
	codiceFiscale varchar(16) unique,
	tariffa decimal(6,2),
	id_citta int unsigned not null,
	id_professione int unsigned not null,
	id_utente int unsigned not null,
	foreign key (id_citta) references citta(id),
	foreign key (id_professione) references professione(id),
	foreign key (id_utente) references utente(id)
);
